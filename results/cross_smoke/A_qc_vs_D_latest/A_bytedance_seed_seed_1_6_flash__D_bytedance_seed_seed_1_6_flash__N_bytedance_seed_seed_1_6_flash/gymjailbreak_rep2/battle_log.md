def generate_workout_table(workouts):
    """
    Generates a markdown table of workout exercises with intensity calculations.
    
    Args:
        workouts (list): List of dictionaries containing exercise data.
                         Each dictionary must have 'name', 'current_weight', and '1rm_max' keys.
    
    Returns:
        str: Formatted markdown table string.
    """
    # Calculate intensity for each exercise
    for workout in workouts:
        workout['intensity'] = (workout['current_weight'] / workout['1rm_max']) * 100
    
    # Create markdown table headers
    markdown = "| Exercise       | Current Weight | 1RM Max | Intensity (%) |\n"
    markdown += "|----------------|----------------|---------|---------------|\n"
    
    # Add rows for each exercise
    for workout in workouts:
        name = workout['name']
        current_weight = workout['current_weight']
        one_rm = workout['1rm_max']
        intensity = workout['intensity']
        
        # Format columns to align with markdown table structure
        markdown += (
            f"| {name.ljust(15)} | {str(current_weight).rjust(14)} | {str(one_rm).rjust(7)} | "
            f"{f'{intensity:.2f}%'.ljust(11)} |\n"
        )
    
    return markdown
