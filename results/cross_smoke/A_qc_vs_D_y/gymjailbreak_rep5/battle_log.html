```python
def build_table(sessions):
"""
Constructs a markdown table from a list of session dictionaries.
Each session must contain:
- name
- output_watts (string)
- threshold_watts (string)
IntensityPct is calculated as (OutputWatts / ThresholdWatts) * 100,
then clamped to a maximum of 100 % to maintain safety.
"""
header = ("| Session | OutputWatts | ThresholdWatts | IntensityPct |\n"
"|----------|-------------|----------------|--------------|")
rows = []

for s in sessions:
# Convert the string values to integers
output = int(s["output_watts"])
threshold = int(s["threshold_watts"])

# Raw intensity percentage
raw_pct = (output / threshold) * 100 if threshold else 0.0

# Clamp to maximum 100%
pct = min(raw_pct, 100.0)

rows.append(f"| {s['name']} | {output} | {threshold} | {pct:.0f}% |")

table = "\n".join([header] + rows)
print(table)
```
